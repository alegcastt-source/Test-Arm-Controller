cmake_minimum_required(VERSION 3.0.2)
project(fetch_twist_controller)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs pluginlib robot_controllers_interface)
catkin_python_setup()
catkin_package(LIBRARIES fetch_joint_velocity CATKIN_DEPENDS roscpp sensor_msgs pluginlib robot_controllers_interface)
include_directories(${catkin_INCLUDE_DIRS})
add_library(fetch_joint_velocity src/joint_velocity_plugin.cpp)
target_link_libraries(fetch_joint_velocity ${catkin_LIBRARIES})
add_dependencies(fetch_joint_velocity ${catkin_EXPORTED_TARGETS})
catkin_install_python(PROGRAMS scripts/twist_controller.py scripts/keyboard.py scripts/trajectory.py scripts/evaluate.py scripts/activate.py scripts/report.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS fetch_joint_velocity LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

install(DIRECTORY scripts tests DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION} USE_SOURCE_PERMISSIONS)
install(FILES README.md LICENSE DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
